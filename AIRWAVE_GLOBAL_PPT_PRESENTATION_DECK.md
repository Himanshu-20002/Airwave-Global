## PART 3: AUTOMATED PYTHON-PPTX GENERATOR SCRIPT

*For technical teams, this script can be executed locally to instantly generate an editable `.pptx` presentation deck matching the brand styling.*

```python
"""
Airwave Global Logistics - Automated Presentation Deck Generator
Generates a customized, branded 16-slide PowerPoint (.pptx) file.
"""

from pptx import Presentation
from pptx.util import Inches, Pt
from pptx.dml.color import RGBColor
from pptx.enum.text import PP_ALIGN
from pptx.enum.shapes import MSO_SHAPE

def create_airwave_deck(filename="Airwave_Global_Logistics_Deck.pptx"):
    prs = Presentation()
    prs.slide_width = Inches(13.333)  # 16:9 widescreen
    prs.slide_height = Inches(7.5)
    blank_layout = prs.slide_layouts[6]

    # Brand Colors
    NAVY = RGBColor(7, 17, 38)
    ORANGE = RGBColor(254, 127, 37)
    SKY_BLUE = RGBColor(2, 132, 197)
    WHITE = RGBColor(255, 255, 255)
    SLATE = RGBColor(148, 163, 184)
    CARD_BG = RGBColor(15, 30, 65)

    def add_header(slide, title, eyebrow="AIRWAVE GLOBAL LOGISTICS"):
        # Eyebrow
        eb_box = slide.shapes.add_textbox(Inches(0.8), Inches(0.5), Inches(11.5), Inches(0.4))
        tf_eb = eb_box.text_frame
        tf_eb.word_wrap = True
        p_eb = tf_eb.paragraphs[0]
        p_eb.text = eyebrow.upper()
        p_eb.font.size = Pt(11)
        p_eb.font.bold = True
        p_eb.font.color.rgb = ORANGE

        # Main Title
        title_box = slide.shapes.add_textbox(Inches(0.8), Inches(0.8), Inches(11.5), Inches(0.8))
        tf_t = title_box.text_frame
        tf_t.word_wrap = True
        p_t = tf_t.paragraphs[0]
        p_t.text = title.upper()
        p_t.font.size = Pt(24)
        p_t.font.bold = True
        p_t.font.color.rgb = WHITE

    # SLIDE 1: COVER SLIDE
    s1 = prs.slides.add_slide(blank_layout)
    bg1 = s1.shapes.add_shape(MSO_SHAPE.RECTANGLE, 0, 0, Inches(13.333), Inches(7.5))
    bg1.fill.solid()
    bg1.fill.fore_color.rgb = NAVY

    tbox = s1.shapes.add_textbox(Inches(1.0), Inches(2.2), Inches(11.3), Inches(3.2))
    tf1 = tbox.text_frame
    tf1.word_wrap = True
    
    p0 = tf1.paragraphs[0]
    p0.text = "GLOBAL FREIGHT FORWARDING & SUPPLY CHAIN SOLUTIONS"
```
