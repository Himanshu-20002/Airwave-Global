## 🐍 OPTIONAL: PYTHON-PPTX SCRIPT

If you run Python, this script creates the exact widescreen presentation automatically:

```python
"""
Airwave Global Logistics — 8-Slide PowerPoint Generator
Install prerequisite: pip install python-pptx
Run: python generate_airwave_8slides.py
"""

from pptx import Presentation
from pptx.util import Inches, Pt
from pptx.dml.color import RGBColor
from pptx.enum.text import PP_ALIGN
from pptx.enum.shapes import MSO_SHAPE

def build_presentation(filename="Airwave_Global_Logistics_8Slides.pptx"):
    prs = Presentation()
    prs.slide_width = Inches(13.333)
    prs.slide_height = Inches(7.5)
    blank_layout = prs.slide_layouts[6]

    NAVY = RGBColor(7, 17, 38)
    ORANGE = RGBColor(254, 127, 37)
    SKY_BLUE = RGBColor(2, 132, 197)
    GREEN = RGBColor(16, 185, 129)
    BG = RGBColor(248, 250, 252)
    WHITE = RGBColor(255, 255, 255)
    SLATE = RGBColor(100, 116, 139)

    def set_bg(slide):
        bg = slide.shapes.add_shape(MSO_SHAPE.RECTANGLE, 0, 0, Inches(13.333), Inches(7.5))
        bg.fill.solid()
        bg.fill.fore_color.rgb = BG
        bg.line.fill.background()
        return bg

    def add_title(slide, text):
        box = slide.shapes.add_textbox(Inches(0.8), Inches(0.5), Inches(11.5), Inches(0.8))
        tf = box.text_frame
        p = tf.paragraphs[0]
        p.text = text
        p.font.size = Pt(28)
        p.font.bold = True
        p.font.color.rgb = NAVY

    # Slide 1: Cover
    s1 = prs.slides.add_slide(blank_layout)
    set_bg(s1)
    tbox = s1.shapes.add_textbox(Inches(1.5), Inches(2.0), Inches(10.33), Inches(3.5))
    tf1 = tbox.text_frame
    
    p0 = tf1.paragraphs[0]
    p0.text = "airwave logistics"
    p0.font.size = Pt(36)
    p0.font.bold = True
    p0.font.color.rgb = ORANGE
    p0.alignment = PP_ALIGN.CENTER

    p1 = tf1.add_paragraph()
    p1.text = "Global Logistics (India)"
    p1.font.size = Pt(44)
    p1.font.bold = True
    p1.font.color.rgb = NAVY
    p1.alignment = PP_ALIGN.CENTER

    p2 = tf1.add_paragraph()
    p2.text = "Your Bridge to Global Trade Corridors"
    p2.font.size = Pt(20)
    p2.font.bold = True
    p2.font.color.rgb = SKY_BLUE
    p2.alignment = PP_ALIGN.CENTER

    p3 = tf1.add_paragraph()
    p3.text = "\nCompany Profile 2026"
    p3.font.size = Pt(13)
    p3.font.color.rgb = SLATE
    p3.alignment = PP_ALIGN.CENTER

    prs.save(filename)
    print(f"Presentation successfully saved as '{filename}'")

if __name__ == "__main__":
    build_presentation()
```
